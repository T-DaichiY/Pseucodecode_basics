20. Local & Global Variables
1Sample Reference
score = 0                          # GLOBAL: created outside any function

def add_points(points):
    global score                   # needed to CHANGE a global variable
    score = score + points

def double(number):
    result = number * 2            # LOCAL: exists only inside this function
    return result

add_points(10)
print(score)                       # 10
print(double(4))                   # 8
# print(result) would cause a NameError - result is local to double()
